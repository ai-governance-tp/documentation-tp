% 3D strana UML uzla
\newcommand\kocka[2]{%
    \filldraw[fill=#2, semithick] (#1.north west) -- ++(0.28,0.28) -- ([xshift=0.28cm, yshift=0.28cm]#1.north east)
        -- ([xshift=0.28cm, yshift=0.28cm]#1.south east) -- (#1.south east) -- (#1.north east) -- cycle;
    \draw[semithick] (#1.north east) -- ++(0.28,0.28);}
\newcommand\st[1]{{\scriptsize\guillemotleft#1\guillemotright}}
